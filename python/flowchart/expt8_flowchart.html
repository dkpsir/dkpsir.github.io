Experiment 8 – Machine Monitoring and Data Analysis using NumPy and Pandas

Experiment 8 – Flow Chart and Python Program
Complete Python Program
import numpy as np
import pandas as pd

# Machine data
data = {
    "Machine": ["M1", "M2", "M3", "M4", "M5"],
    "Speed_RPM": [1200, 1500, 1800, 1400, 1600],
    "Temperature_C": [65, 72, 85, 68, 78],
    "Vibration_mm_s": [2.1, 2.8, 4.5, 2.4, 3.2],
    "Power_kW": [5.2, 6.5, 8.0, 5.8, 7.1]
}

# Create DataFrame
df = pd.DataFrame(data)

print("Machine Data:")
print(df)

# Calculate average values
average_temperature = np.mean(df["Temperature_C"])
average_vibration = np.mean(df["Vibration_mm_s"])
average_power = np.mean(df["Power_kW"])

print("\nAverage Temperature:",
      round(average_temperature, 2), "°C")
print("Average Vibration:",
      round(average_vibration, 2), "mm/s")
print("Average Power:",
      round(average_power, 2), "kW")

# Find machine with maximum vibration
max_vibration = df["Vibration_mm_s"].max()

machine = df.loc[
    df["Vibration_mm_s"].idxmax(),
    "Machine"
]

print("\nMaximum Vibration:",
      max_vibration, "mm/s")
print("Machine with Maximum Vibration:", machine)

# Classify machine condition
df["Condition"] = np.where(
    df["Vibration_mm_s"] > 3.0,
    "Needs Inspection",
    "Normal"
)

print("\nMachine Condition:")
print(df[["Machine", "Vibration_mm_s", "Condition"]])
