# Experiment 5: Material Data Base Using Dictionary

def display(material):
    for key, value in material.items():
        print(key, ":", value)

def add_property(material):
    key = input("Enter property name : ")
    value = input("Enter property value : ")
    material[key] = value

def delete_property(material):
    key = input("Enter property to delete : ")
    if key in material:
        del material[key]
    else:
        print("Property Not Found.")

material = {
    "name": "Mild Steel",
    "density": 7850,
    "hardness": 150,
    "youngs_modulus": 200000,
    "poisson_ratio": 0.30
}

display(material)

choice = int(input("\nEnter your choice (1-Add, 2-Delete): "))

if choice == 1:
    add_property(material)
    display(material)
elif choice == 2:
    delete_property(material)
    display(material)
else:
    print("Invalid Choice.")
