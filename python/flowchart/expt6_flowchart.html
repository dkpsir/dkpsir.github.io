Complete Python Program
import csv

# 1. String methods
name = "  gear  "

print("Original:", name)
print("Uppercase:", name.upper())
print("Lowercase:", name.lower())
print("Without spaces:", name.strip())
print("Length:", len(name.strip()))
print("Replaced:", name.replace("gear", "shaft"))
print("Count of 'a':", name.count("a"))

# 2. Write to a text file
with open("data.txt", "w") as file:
    file.write("Gear\n")
    file.write("Steel\n")
    file.write("120 mm")

# 3. Read from text file
with open("data.txt", "r") as file:
    print("\nText file data:")
    print(file.read())

# 4. Write to CSV file
with open("data.csv", "w", newline="") as file:
    writer = csv.writer(file)
    writer.writerow(["Name", "Material", "Diameter"])
    writer.writerow(["Gear", "Steel", "120 mm"])

print("\nCSV file created successfully.")
