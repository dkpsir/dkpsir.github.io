# Experiment 7: Machine Temperature Monitoring

class TemperatureError(Exception):
    """Custom exception for unsafe machine temperature."""
    pass

try:
    temperature = float(input("Enter machine temperature (°C): "))
    if temperature < 0 or temperature > 100:
        raise TemperatureError("Machine temperature is outside the safe range!")

except ValueError:
    print("Error: Please enter a valid numerical temperature.")

except TemperatureError as e:
    print("Temperature Error:", e)

else:
    print("Machine temperature is safe.")
    print("Temperature:", temperature, "°C")

finally:
    print("Temperature checking completed.")
