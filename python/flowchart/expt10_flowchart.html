Experiment 10 – Heat Exchanger Performance Analysis using Pandas and Seaborn

Experiment 10 – Flow Chart and Python Program
Complete Python Program
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# Step 1: Experimental heat exchanger data
data = {
    "Test": ["Test 1", "Test 2", "Test 3", "Test 4", "Test 5"],
    "Hot_Inlet_C": [80, 85, 90, 95, 100],
    "Hot_Outlet_C": [65, 68, 70, 73, 75],
    "Cold_Inlet_C": [25, 27, 28, 30, 32],
    "Cold_Outlet_C": [40, 43, 46, 49, 52]
}

df = pd.DataFrame(data)

# Step 2: Define operating parameters
mass_flow_rate = 0.1      # kg/s
specific_heat = 4180      # J/(kg degree C)

# Step 3: Calculate heat transfer rate on hot side
df["Heat_Transfer_kW"] = mass_flow_rate * specific_heat * (df["Hot_Inlet_C"] - df["Hot_Outlet_C"]) / 1000

# Step 4: Calculate heat capacity rates
hot_capacity = mass_flow_rate * specific_heat
cold_capacity = mass_flow_rate * specific_heat

# Maximum possible heat transfer
df["Q_Max_kW"] = min(hot_capacity, cold_capacity) * (df["Hot_Inlet_C"] - df["Cold_Inlet_C"]) / 1000

# Step 5: Calculate effectiveness
df["Effectiveness_pct"] = df["Heat_Transfer_kW"] / df["Q_Max_kW"] * 100

# Step 6: Display the results
print("HEAT EXCHANGER PERFORMANCE")
print(df.round(2).to_string(index=False))

print("\nAverage Heat Transfer Rate:", round(df["Heat_Transfer_kW"].mean(), 2), "kW")
print("Average Effectiveness:", round(df["Effectiveness_pct"].mean(), 2), "%")

# Step 7: Plot heat transfer rate
plt.figure(figsize=(7, 4))
sns.barplot(data=df, x="Test", y="Heat_Transfer_kW")
plt.title("Heat Transfer Rate for Each Test")
plt.xlabel("Experimental Test")
plt.ylabel("Heat Transfer Rate (kW)")
plt.tight_layout()
plt.show()

# Step 8: Plot hot and cold outlet temperatures
plt.figure(figsize=(7, 4))
plt.plot(df["Test"], df["Hot_Outlet_C"], marker="o", label="Hot outlet")
plt.plot(df["Test"], df["Cold_Outlet_C"], marker="s", label="Cold outlet")
plt.title("Outlet Temperature Comparison")
plt.xlabel("Experimental Test")
plt.ylabel("Temperature (°C)")
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.show()

# Step 9: Correlation analysis
columns = ["Hot_Inlet_C", "Hot_Outlet_C", "Cold_Inlet_C", "Cold_Outlet_C", "Heat_Transfer_kW", "Effectiveness_pct"]

plt.figure(figsize=(9, 6))
sns.heatmap(df[columns].corr(), annot=True, cmap="coolwarm", vmin=-1, vmax=1)
plt.title("Heat Exchanger Parameter Correlations")
plt.tight_layout()
plt.show()
