# Program: Stress and Strain Calculator Using Functions and While Loop
import math

# Function to Display Results
def display(area, stress_value, long_strain, lat_strain):
    print("\n----------- RESULTS -----------")
    print("Area =", round(area, 2), "mm²")
    print("Stress =", round(stress_value, 2), "N/mm²")
    print("Longitudinal Strain =", round(long_strain, 6))
    print("Lateral Strain =", round(lat_strain, 6))

# Function to Calculate Stress
def stress(load, area):
    sigma = load / area
    return sigma

# Function to Calculate Longitudinal Strain
def longitudinal_strain(load, length, area, young_modulus):
    long_strain = (load * length) / (area * young_modulus)
    return long_strain

# Function to Calculate Lateral Strain
def lateral_strain(change, original):
    lat_strain = change / original
    return lat_strain

# Function for Circular Cross Section
def circular_section():
    print("\nCIRCULAR CROSS SECTION")
    d = float(input("Enter Diameter (mm): "))
    L = float(input("Enter Length (mm): "))
    P = float(input("Enter Tensile Load (N): "))
    E = float(input("Enter Young's Modulus (N/mm²): "))
    delta_d = float(input("Enter Change in Diameter (mm): "))
    
    area = math.pi * d * d / 4
    s = stress(P, area)
    long_strain = longitudinal_strain(P, L, area, E)
    lat_strain = lateral_strain(delta_d, d)
    
    display(area, s, long_strain, lat_strain)

# Function for Rectangular Cross Section
def rectangular_section():
    print("\nRECTANGULAR CROSS SECTION")
    b = float(input("Enter Width (mm): "))
    t = float(input("Enter Thickness (mm): "))
    L = float(input("Enter Length (mm): "))
    P = float(input("Enter Tensile Load (N): "))
    E = float(input("Enter Young's Modulus (N/mm²): "))
    delta_b = float(input("Enter Change in Width (mm): "))
    
    area = b * t
    s = stress(P, area)
    long_strain = longitudinal_strain(P, L, area, E)
    lat_strain = lateral_strain(delta_b, b)
    
    display(area, s, long_strain, lat_strain)

# Main Program
while True:
    print("\nSTRESS AND STRAIN CALCULATOR")
    print("1. Circular Cross Section")
    print("2. Rectangular Cross Section")
    print("3. Exit")
    
    choice = int(input("Enter your choice (1, 2 or 3): "))
    
    if choice == 1:
        circular_section()
    elif choice == 2:
        rectangular_section()
    elif choice == 3:
        print("\nProgram Terminated.")
        break
    else:
        print("\nInvalid Choice.")
    
    again = input("\nDo you want to calculate another structure? (Y/N): ")
    if again.upper() != "Y":
        print("\nThank You.")
        break
