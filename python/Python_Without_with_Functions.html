WITH FUNCTIONS

def get_input():
    length = float(input("Enter Length : "))
    breadth = float(input("Enter Breadth : "))
    return length, breadth
def area(length, breadth):
    return length * breadth
def perimeter(length, breadth):
    return 2 * (length + breadth)
def moment_of_inertia(length, breadth):
    return (breadth * (length ** 3)) / 12
def display(a, p, m):
    print("Area =", a)
    print("Perimeter =", p)
    print("Moment of Inertia =", m)
# Calculation for First Pair
length, breadth = get_input()
display(area(length, breadth),perimeter(length, breadth),moment_of_inertia(length, breadth))
