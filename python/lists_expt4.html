def display(steel_grade, steel_type, applications):
    for i in range(len(steel_grade)):
        print(i+1,". Steel Grade :", steel_grade[i],"Steel Type  :", steel_type[i],"Application :", applications[i])
steel_grade = ["AISI 1018", "AISI 1045", "AISI 1095"]
steel_type = ["Low Carbon Steel", "Medium Carbon Steel", "High Carbon Steel"]
applications = ["Bolts and Shafts", "Gears", "Springs"]
display(steel_grade, steel_type, applications)
choice = int(input("\nEnter your choice (1-Add, 2-Delete): "))
if choice == 1:
    g = input("Enter Steel Grade : ")
    s = input("Enter Steel Type : ")
    a = input("Enter Application : ")
    steel_grade.append(g)
    steel_type.append(s)
    applications.append(a)
    display(steel_grade, steel_type, applications)
elif choice == 2:
    grade = input("Enter Steel Grade to Delete : ")
    if grade in steel_grade:
        index = steel_grade.index(grade)
        steel_grade.pop(index)
        steel_type.pop(index)
        applications.pop(index)
        display(steel_grade, steel_type, applications)
    else:
        print("\nSteel Grade Not Found.")
else:
    print("\nInvalid Choice.")
