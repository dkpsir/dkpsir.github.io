p = float(input("Principal amount: "))
t = float(input("Time period (years): "))
r = float(input("Rate of interest (%): "))

si = (p * t * r) / 100
final_amount = p + si

print("\n---  INTEREST DETAILS ---")
print(f"Simple Interest = {si:.2f}")
print(f"Final Amount    = {final_amount:.2f}")
