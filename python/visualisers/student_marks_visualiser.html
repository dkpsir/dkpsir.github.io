import numpy as np
import pandas as pd

# Marks of five students in three subjects
data = {
    "Name": ["Anu", "Balu", "Chandu", "Divya", "Eswar"],
    "Python": [85, 72, 90, 65, 78],
    "Statistics": [80, 75, 88, 70, 82],
    "Data Science": [90, 68, 92, 75, 85]
}

# Create a DataFrame
df = pd.DataFrame(data)

# Calculate total and average marks
marks = df[["Python", "Statistics", "Data Science"]]
df["Total"] = np.sum(marks, axis=1)
df["Average"] = np.mean(marks, axis=1)

# Assign result based on average
df["Result"] = np.where(
    df["Average"] >= 70, "Pass", "Needs Improvement"
)

# Display results
print("STUDENT PERFORMANCE REPORT")
print(df.round(2).to_string(index=False))

# Find the highest average
topper = df.loc[df["Average"].idxmax()]

print("\nTop Performer:", topper["Name"])
print("Highest Average:", round(topper["Average"], 2))
