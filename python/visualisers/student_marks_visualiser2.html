import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# Step 1: Create a dataset
data = {
    "Student": ["Anu", "Balu", "Chandu", "Divya", "Eswar"],
    "Python": [85, 72, 90, 65, 78],
    "Statistics": [80, 75, 88, 70, 82],
    "DataScience": [90, 68, 92, 75, 85]
}

df = pd.DataFrame(data)

# Step 2: Calculate total and average
subjects = ["Python", "Statistics", "DataScience"]

df["Total"] = df[subjects].sum(axis=1)
df["Average"] = df[subjects].mean(axis=1)

print("STUDENT DATA")
print(df.to_string(index=False))

# Step 3: Display descriptive statistics
print("\nSTATISTICAL SUMMARY")
print(df[subjects].describe().round(2))

# Step 4: Visualize subject-wise average marks
subject_averages = df[subjects].mean()

plt.figure(figsize=(7, 4))
subject_averages.plot(kind="bar")
plt.title("Average Marks by Subject")
plt.xlabel("Subject")
plt.ylabel("Average Marks")
plt.xticks(rotation=0)
plt.tight_layout()
plt.show()

# Step 5: Correlation heatmap
plt.figure(figsize=(6, 4))
sns.heatmap(
    df[subjects].corr(),
    annot=True,
    cmap="coolwarm",
    vmin=-1,
    vmax=1
)
plt.title("Correlation Between Subject Marks")
plt.tight_layout()
plt.show()

# Step 6: Identify the highest scorer
topper = df.loc[df["Average"].idxmax()]
print("\nTop Performer:", topper["Student"])
print("Average Marks:", round(topper["Average"], 2))
